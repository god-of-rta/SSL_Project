\begin{frame}{Architecture Overview (P1)}
\begin{itemize}
    \item \textbf{End-to-End Pipeline}: Collector (Bash) $\to$ Cleaner (Python) $\to$ Analyzer (Python) $\to$ Alerts (Python) $\to$ Web UI (Flask + JS).
    \item One-command launch via \texttt{./sysmon} with automatic virtual environment initialization.
    \item Fully local execution at \texttt{127.0.0.1:5000} without external cloud dependencies.
\end{itemize}
\end{frame}

\begin{frame}{Backend Server \& API Design (P4)}
\begin{itemize}
    \item Flask-powered REST endpoints on \texttt{127.0.0.1}.
    \item Safe process control via \texttt{start.sh}, \texttt{stop.sh}, and \texttt{status.sh}.
    \item Adaptive downsampling for historical metrics (1h raw, 24h 1-minute, 7d 10-minute).
\end{itemize}
\end{frame}

\begin{frame}{Frontend Dashboard (P3)}
\begin{itemize}
    \item Pure vanilla JavaScript and HTML5/CSS3 (dark/light themes).
    \item Offline-bundled Chart.js for real-time visualization.
    \item Real-time alert toasts, process tracking tables, and live settings management.
\end{itemize}
\end{frame}

\begin{frame}{Alert Engine \& Performance Profiling (P5)}
\begin{itemize}
    \item State machine alerts with sustained breach verification and cooldown timers.
    \item \texttt{cProfile} analysis identifying pure-Python cleaning bottlenecks.
    \item Vectorized Pandas cleaner optimization achieving significant speedups on 1M+ rows.
\end{itemize}
\end{frame}
