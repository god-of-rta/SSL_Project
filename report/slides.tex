\section{Introduction and Architecture}

\begin{frame}{SysMon: Motivation, Goals, and Architecture}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{The Core Problem:} Existing Unix monitors are either too simplistic (command-line only, no history or alerts) or too heavy (Prometheus/Datadog consume 200MB+ RAM and distort CPU measurements).
        \item \textbf{Our Primary Goal:} Deliver an autonomous, zero-configuration monitoring suite with less CPU probe overhead, using local CSV/JSON files with no external database dependencies.
        \item \textbf{Four-Stage Decoupled Pipeline:}
        \begin{itemize}
            \setlength{\itemsep}{3pt}
            \item \textit{Collector (Bash):} Polls 16 kernel metrics from \texttt{/proc} \& \texttt{/sys} every 5 seconds.
            \item \textit{Cleaner (Python):} Validates ranges, forward-fills brief gaps, and tags MAD outliers.
            \item \textit{Analyzer (Python):} Computes multi-window stats, linear disk forecasts, and Z-scores.
            \item \textit{Alerts \& Web UI (Flask):} Tracks sustained rules and serves a real-time streaming dashboard.
        \end{itemize}
        \item \textbf{Scope \& Roles:} Varun - Analysis and system integration,
        Harsha - Bash scripts,
        Prerit - Frontend,
        Amogh - Backend,
        Amulya - Alerts and profiling,
        Amitha - Bash scripts and Latex files
    \end{itemize}
\end{frame}

\begin{frame}{System Telemetry Collector (\texttt{bin/collector.sh})}{P2: Harsha --- Telemetry Collection \& Daemon Management}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Lightweight Data Gathering:}
\begin{itemize}
    \item \textbf{Low Overhead}: Reads system files (\texttt{/proc}, \texttt{/sys}) directly instead of heavy tools ($<0.05\%$ CPU).
    \item \textbf{16 Key Metrics}: CPU usage deltas, true available memory, disk, network rates, and temperatures.
    \item \textbf{Smart GPU Monitoring}: Skips polling if dedicated graphics card is asleep, saving battery.
    \item \textbf{Process Tracking}: Records top 5 CPU processes while filtering brief startup glitches.
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Reliable Daemon Lifecycle:}
\begin{itemize}
    \item \textbf{Simple Controls}: Managed cleanly via \texttt{start.sh}, \texttt{stop.sh}, and \texttt{status.sh}.
    \item \textbf{Single-Instance Lock}: Prevents duplicate collectors from running concurrently.
    \item \textbf{Clean Shutdown}: Gracefully flushes data and closes files on exit.
\end{itemize}

\vspace{0.3em}
\begin{block}{Interface Contract}
Generates daily 16-column CSVs and hardware summary (\texttt{system\_info.json}).
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Data Cleaning Pipeline (\texttt{src/cleaner.py})}{P2: Harsha --- Data Quality, Imputation \& Retention}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Automated Data Sanitization:}
\begin{itemize}
    \item \textbf{Validation}: Checks realistic bounds (0--100\%, temperatures $\le 120^\circ\text{C}$); drops corrupted rows.
    \item \textbf{Sorting \& Deduplication}: Orders rows by timestamp and discards duplicate reads.
    \item \textbf{Handling Missing Data}:
    \begin{itemize}
        \item \emph{Short pauses ($1$--$3$ cycles)}: Forward-fills prior values to keep trends smooth.
        \item \emph{Long gaps ($>3$ cycles)}: Flags sleep/downtime without making up fake data.
    \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Spike Protection \& Maintenance:}
\begin{itemize}
    \item \textbf{Highlights Sudden Spikes}: Flags unusual jumps using a rolling-median window.
    \item \textbf{Never Deletes Spikes}: Keeps real load spikes intact so alerts still trigger!
    \item \textbf{Automatic Disk Cleanup}: Deletes files older than 7 days (\texttt{retention\_hours}).
\end{itemize}

\vspace{0.3em}
\begin{block}{Verification}
$100\%$ automated test pass rate (\texttt{test\_collector.py} \& \texttt{test\_cleaner.py}).
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Statistical Analysis and Pipeline Automation}

\begin{frame}{Statistical Analysis Engine (\texttt{src/analyzer.py})}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{Multi-Window Rolling Statistics:} Computes mean, min, max, 95th percentile ($p_{95}$ via linear rank interpolation), and sample standard deviation ($ddof=1$) across 4 windows (\texttt{1h}, \texttt{24h}, \texttt{7d}, \texttt{all}).
        \item \textbf{Predictive Disk-Full Forecast:} Fits Ordinary Least Squares (OLS) linear regression on root disk percentage over time to compute daily consumption rate ($\Delta\%/\text{day}$) and projected days to 100\% capacity.
        \item \textbf{Z-Score Anomaly Detection:} Flags operational spikes where $|z| = \frac{|x - \mu|}{\sigma} > 3.0$ standard deviations from the window mean, identifying rare statistical outliers.
        \item \textbf{Core-Normalized Process Ranking:} Scales raw \texttt{ps} CPU utilization by physical core count ($\min(100.0, \frac{\text{avg\_cpu}}{\text{cpu\_cores}})$), preventing confusing $>1000\%$ readings and matching desktop resource monitors.
    \end{itemize}
\end{frame}

\begin{frame}{Pipeline Runner and Optimization (\texttt{bin/run\_pipeline.sh})}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{Sequential Stage Orchestration:} Executes Cleaner $\rightarrow$ Analyzer $\rightarrow$ Alerts in strict sequence, halting immediately if any stage fails to isolate downstream errors.
        \item \textbf{Smart Change-Detection Caching:} Hashes raw CSV sizes and mtimes in \texttt{.pipeline\_raw\_stamp}; skips cleaner and analyzer when data is static.
        \item \textbf{Subprocess Thread Pool Clamping:} Explicitly sets \texttt{OPENBLAS\_NUM\_THREADS=1} and \texttt{OMP\_NUM\_THREADS=1} to prevent short Python subprocesses from spawning 32 threads and churning the CPU.
    \end{itemize}
\end{frame}

\begin{frame}{SysMon Launcher and Real Telemetry Findings}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{One-Click System Launcher (\texttt{./sysmon}):} Single root command creates \texttt{.venv}, installs requirements, initializes configs (\texttt{--init-defaults}), starts the server, and automatically opens the browser.
        \item \textbf{Real Hardware Telemetry Dataset:} Validated against 4,814 clean rows and 23,886 process records collected over 2 days.
        \item \textbf{Automated Alert Verification:} Alert engine fired warning at 80$^\circ$C and critical at 90$^\circ$C under sustained load, automatically resolving once temperature cooled to 77$^\circ$C.
    \end{itemize}
\end{frame}

\section{Backend Performance and Operations}

% ------------------------------------------------------------
\begin{frame}{Performance Problems \& Solutions}
\footnotesize
\begin{tabularx}{\linewidth}{@{}L{0.34\linewidth}Y@{}}
\toprule
\textbf{Problem} & \textbf{Solution / outcome} \\
\midrule
Repeated clicks / overlapping pipeline runs
& \texttt{threading.Lock}: one active run; reject or skip overlap. \\
\addlinespace[1mm]
Scheduled work inside request handling
& Daemon background thread: periodic pipeline runs outside the request handler. \\
\addlinespace[1mm]
Excess CPU threads from numerical libraries
& OpenBLAS / OMP / MKL limited to \texttt{1} before Pandas and NumPy imports. \\
\addlinespace[1mm]
Partially written JSON or config files
& Atomic write: temporary file $\rightarrow$ \texttt{os.replace()} $\rightarrow$ complete-file replacement. \\
\addlinespace[1mm]
Manual row-by-row CSV processing
& Pandas: fast table operations for filtering, sorting, conversion, and resampling. \texttt{open()} remains for simple file reads. \\
\addlinespace[1mm]
Too many chart points / dashboard slowdown
& Downsampling: fewer points for longer ranges; smaller responses and lighter chart rendering. \\
\bottomrule
\end{tabularx}

\vspace{2mm}
\scriptsize
\textbf{Bucket sizes:} \texttt{1h} = 5 sec samples \quad
\texttt{24h} = 1 min buckets \quad
\texttt{7d} = 10 min buckets.\\
\textbf{Bucket rule:} numeric values = average; flags = maximum; timestamp = last.\\
\textit{Note:} current \texttt{read\_csv()} loads the clean CSV before downsampling. Downsampling reduces response/browser load, not the initial file-load memory.
\end{frame}

\section{Integration and Profiling}

\begin{frame}{Performance Profiling and Hardware Optimizations}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{Eliminating Dedicated GPU Wakeups:} Checked PCI power state via sysfs to bypass \texttt{nvidia-smi} when the dGPU is asleep in D3cold, slashing polling latency from \textbf{520 ms to 18 ms} ($28.8\times$ faster).
        \item \textbf{Significant CPU Overhead Reduction:} Collector CPU dropped from $1.85\% \to \mathbf{0.32\%}$ at idle and $4.10\% \to \mathbf{0.74\%}$ under heavy stress, easily meeting our $<1\%$ probe budget.
        \item \textbf{Lean Memory Footprint:} Collector runs in only \textbf{4.2 MB} of RAM and the Flask server uses \textbf{28.4 MB}, remaining well below our 50 MB design limit.
    \end{itemize}
\end{frame}
