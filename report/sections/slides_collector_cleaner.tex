\begin{frame}{System Telemetry Collector (\texttt{bin/collector.sh})}{P2: Harsha --- Telemetry Collection \& Daemon Management}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Lightweight Data Gathering:}
\begin{itemize}
    \item \textbf{Low Overhead}: Reads system files (\texttt{/proc}, \texttt{/sys}) directly instead of heavy tools ($<0.05\%$ CPU).
    \item \textbf{16 Key Metrics}: CPU usage deltas, true available memory, disk, network rates, and temperatures.
    \item \textbf{Smart GPU Monitoring}: Skips polling if dedicated graphics card is asleep, saving battery.
    \item \textbf{Process Tracking}: Records top 5 CPU processes while filtering brief startup glitches.
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Reliable Daemon Lifecycle:}
\begin{itemize}
    \item \textbf{Simple Controls}: Managed cleanly via \texttt{start.sh}, \texttt{stop.sh}, and \texttt{status.sh}.
    \item \textbf{Single-Instance Lock}: Prevents duplicate collectors from running concurrently.
    \item \textbf{Clean Shutdown}: Gracefully flushes data and closes files on exit.
\end{itemize}

\vspace{0.3em}
\begin{block}{Interface Contract}
Generates daily 16-column CSVs and hardware summary (\texttt{system\_info.json}).
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Data Cleaning Pipeline (\texttt{src/cleaner.py})}{P2: Harsha --- Data Quality, Imputation \& Retention}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Automated Data Sanitization:}
\begin{itemize}
    \item \textbf{Validation}: Checks realistic bounds (0--100\%, temperatures $\le 120^\circ\text{C}$); drops corrupted rows.
    \item \textbf{Sorting \& Deduplication}: Orders rows by timestamp and discards duplicate reads.
    \item \textbf{Handling Missing Data}:
    \begin{itemize}
        \item \emph{Short pauses ($1$--$3$ cycles)}: Forward-fills prior values to keep trends smooth.
        \item \emph{Long gaps ($>3$ cycles)}: Flags sleep/downtime without making up fake data.
    \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Spike Protection \& Maintenance:}
\begin{itemize}
    \item \textbf{Highlights Sudden Spikes}: Flags unusual jumps using a rolling-median window.
    \item \textbf{Never Deletes Spikes}: Keeps real load spikes intact so alerts still trigger!
    \item \textbf{Automatic Disk Cleanup}: Deletes files older than 7 days (\texttt{retention\_hours}).
\end{itemize}

\vspace{0.3em}
\begin{block}{Verification}
$100\%$ automated test pass rate (\texttt{test\_collector.py} \& \texttt{test\_cleaner.py}).
\end{block}
\end{column}
\end{columns}
\end{frame}
